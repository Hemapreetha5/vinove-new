lamp new:
# 1. Update system
sudo apt update && sudo apt full-upgrade -y

# 2. Install Apache2
sudo apt install apache2 apache2-utils -y
sudo systemctl enable apache2 --now
sudo systemctl status apache2 --no-pager

# 3. Install MariaDB (MySQL drop-in replacement)
sudo apt install mariadb-server mariadb-client -y
sudo systemctl enable mariadb --now
sudo systemctl status mariadb --no-pager

# 4. Secure MariaDB installation
sudo mysql_secure_installation
# Press Enter for current password (none)
# Set root password? → Y
# Enter strong password twice
# Remove anonymous users? → Y
# Disallow root login remotely? → Y
# Remove test database? → Y
# Reload privilege tables? → Y

# 5. Test MariaDB login
sudo mysql -u root -p   # enter the password you just set

# Inside MariaDB prompt:
SHOW DATABASES;
EXIT;

# 6. Add ondrej/php PPA (latest PHP versions)
sudo apt install software-properties-common -y
sudo add-apt-repository ppa:ondrej/php -y
sudo apt update

# 7. Install PHP 8.2 + common extensions (or use 8.3 if preferred)
sudo apt install php8.2 libapache2-mod-php8.2 php8.2-mysql \
                 php8.2-curl php8.2-mbstring php8.2-xml php8.2-zip \
                 php8.2-gd php8.2-bcmath php8.2-intl -y

# 8. Enable PHP module & restart Apache
sudo a2enmod php8.2
sudo systemctl restart apache2

# 9. Verify versions
apache2 -v
php --version
mariadb --version

# 10. Create test PHP file
sudo bash -c 'echo "<?php phpinfo(); ?>" > /var/www/html/info.php'

# 11. Optional: Replace default index page
sudo rm /var/www/html/index.html
sudo bash -c 'echo "<h1>LAMP Stack Ready!</h1><p>Server: $(hostname) @ $(hostname -I)</p>" > /var/www/html/index.html'

# 12. Fix permissions (optional but recommended)
sudo chown -R www-data:www-data /var/www/html
sudo chmod -R 755 /var/www/html
